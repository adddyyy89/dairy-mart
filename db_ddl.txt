CREATE TABLE public."State"
(
    "stateId" integer NOT NULL,
    "stateName" text NOT NULL,
    "countryId" integer NOT NULL,
    CONSTRAINT "STATE_PK" PRIMARY KEY ("stateId"),
    CONSTRAINT "STATE_COUNTRY_FK" FOREIGN KEY ("countryId")
        REFERENCES public."Country" ("countryId") MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
        NOT VALID
);

ALTER TABLE IF EXISTS public."State"
    OWNER to postgres;

------------

CREATE TABLE public."User"
(

    CONSTRAINT "USER_USERADDRESS_FK" FOREIGN KEY ("addressId")
